// 將最多 4 顆不重複號碼，依數值落點分配至 5 柱中的獨立柱位（每柱最多 1 顆） function assignTo5Pillars(nums) { const sorted = [...new Set(nums)].sort((a, b) => a - b).slice(0, 4); const cols = [[], [], [], [], []]; if (sorted.length === 0) return cols; const centers = [6.5, 13.5, 20.0, 26.5, 33.5]; let bestAssign = null; let bestDist = Infinity; function choose(start, chosen) { if (chosen.length === sorted.length) { let dist = 0; for (let i = 0; i < sorted.length; i++) { dist += Math.abs(sorted[i] - centers[chosen[i]]); } if (dist < bestDist) { bestDist = dist; bestAssign = [...chosen]; } return; } for (let i = start; i < 5; i++) { chosen.push(i); choose(i + 1, chosen); chosen.pop(); } } choose(0, []); for (let i = 0; i < sorted.length; i++) { cols[bestAssign[i]].push(sorted[i]); } return cols; } // 第三輪量化模型：多因子定位共識 + 近50期Lag1~3單號拖牌 + 最多4碼與SKIP風控 function getPredictions(data, matrix1, matrix2) { const total = data.length; const preds = {}; const MAX_BALLS = 4; // 硬限制：每期最多 4 碼 const MIN_BALLS = 3; // 風控：少於 3 顆達標觸發 SKIP const MIN_SCORE = 5.0; // 高共識門檻 const W_DRAG = 1.0; // 近 50 期拖牌權重 const DRAG_WINDOW = 50; for (let offset = total - 1; offset >= 0; offset--) { const targetIdx = total - offset; if (targetIdx >= total && offset !== 0) continue; const isFuture = (offset === 0); const periodKey = isFuture ? "NEXT" : data[targetIdx].period; const endIdx = isFuture ? total : targetIdx; const startIdx = Math.max(1, endIdx - 20); if (endIdx < 2) { preds[periodKey] = [[], [], [], [], []]; continue; } // 嚴格只取 0 .. endIdx-1 歷史資料（防資料洩漏） const historySlice = data.slice(0, endIdx); const draws = historySlice.map(h => h.nums.map(Number)); const lastNums = draws[draws.length - 1]; const lastTails = lastNums.map(n => n % 10); // 1. 計算遺漏值 (Omission) const omission = {}; for (let num = 1; num <= 39; num++) { let gap = 0; for (let i = draws.length - 1; i >= 0; i--) { if (draws[i].includes(num)) break; gap++; } omission[num] = gap; } // 2. 計算各位置歷史均值與標準差 const posMean = []; const posStd = []; for (let c = 0; c < 5; c++) { const vals = draws.map(d => d[c]); const mean = vals.reduce((a, b) => a + b, 0) / vals.length; const variance = vals.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / vals.length; posMean.push(mean); posStd.push(Math.max(2.0, Math.sqrt(variance))); } // 3. 近 15 期尾數熱度 const tailFreq = Array(10).fill(0); draws.slice(Math.max(0, draws.length - 15)).forEach(d => { d.forEach(n => tailFreq[n % 10]++); }); // 4. 近 50 期單號隔期拖牌矩陣 (Lag 1 ~ 3) const hDrag = draws.slice(Math.max(0, draws.length - DRAG_WINDOW)); const nDrag = hDrag.length; const dragRaw = Array(40).fill(0); for (let lag = 1; lag <= 3; lag++) { if (nDrag <= lag) continue; const trigCnt = Array(40).fill(0); const coCnt = Array.from({ length: 40 }, () => Array(40).fill(0)); for (let i = lag; i < nDrag; i++) { for (const a of hDrag[i - lag]) { trigCnt[a]++; for (const b of hDrag[i]) coCnt[a][b]++; } } for (const a of hDrag[nDrag - lag]) { if (trigCnt[a] >= 3) { for (let b = 1; b <= 39; b++) { dragRaw[b] += ((coCnt[a][b] + 0.5) / (trigCnt[a] + 3.9)) / lag; } } } } const dragVals = dragRaw.slice(1); const dMin = Math.min(...dragVals); const dMax = Math.max(...dragVals); const dragNorm = Array(40).fill(0); if (dMax > dMin) { for (let b = 1; b <= 39; b++) { dragNorm[b] = (dragRaw[b] - dMin) / (dMax - dMin); } } // 5. 逐柱評分，每柱取最高分候選 const bestPerPillar = []; for (let c = 0; c < 5; c++) { const getCandidates = (matrix, type) => { const counts = {}; for (let i = startIdx; i < endIdx; i++) { matrix[i][c].forEach(v => { if (matrix.slice(startIdx, i).some(m => m[c] && m[c].includes(v))) { counts[v] = (counts[v] || 0) + 1; } }); } const hot = Object.keys(counts) .sort((a, b) => (counts[b] - counts[a]) || (Number(a) - Number(b))) .slice(0, 3) .map(Number); let base = 0; if (type === 1) base = lastNums[c]; else for (let k = 0; k <= c; k++) base += lastTails[k]; const set = new Set(); hot.forEach(d => { [base + d, Math.abs(base - d)].forEach(n => { if (n >= 1 && n <= 39) set.add(n); }); }); return set; }; const s1 = getCandidates(matrix1, 1); const s2 = getCandidates(matrix2, 2); const unionSet = new Set([...s1, ...s2]); const scored = []; unionSet.forEach(num => { const z = Math.abs(num - posMean[c]) / posStd[c]; if (z > 1.6) return; let score = (s1.has(num) && s2.has(num)) ? 3.0 : 1.0; score += Math.max(0, 1.5 - z); const g = omission[num]; if ((g >= 1 && g <= 5) || (g >= 10 && g <= 18)) score += 1.0; if (lastNums.some(ln => Math.abs(num - ln) <= 1)) score += 0.8; score += tailFreq[num % 10] * 0.1; score += W_DRAG * dragNorm[num]; if (score >= MIN_SCORE) { scored.push({ num, score }); } }); scored.sort((a, b) => (b.score - a.score) || (a.num - b.num)); if (scored.length > 0) { bestPerPillar.push(scored[0]); } } // 6. 全盤排序去重，硬性截斷最多 4 碼，不足 3 碼觸發 SKIP bestPerPillar.sort((a, b) => (b.score - a.score) || (a.num - b.num)); const picked = []; const seen = new Set(); for (const item of bestPerPillar) { if (!seen.has(item.num)) { seen.add(item.num); picked.push(item.num); } if (picked.length >= MAX_BALLS) break; } preds[periodKey] = (picked.length >= MIN_BALLS) ? assignTo5Pillars(picked) : [[], [], [], [], []]; } return preds; }
